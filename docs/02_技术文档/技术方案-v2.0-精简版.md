# Moark聚合应用-精简技术方案

## 1. 项目定位

### 1.1 核心原则
- **零服务器成本**：完全本地化应用，无需后端服务器
- **数据主权**：所有数据存储在用户本地，用户拥有绝对控制权
- **隐私保护**：敏感信息本地加密，不上传任何用户数据
- **自动更新**：应用和API接口支持自动更新机制
- **离线能力**：核心功能支持离线使用

### 1.2 目标用户
1. **开发者**：需要快速测试和集成Moark API
2. **个人用户**：需要特定AI功能但注重隐私
3. **中小企业**：需要AI能力但预算有限
4. **研究人员**：需要批量处理和分析AI模型输出

```typescript
// 本地缓存实现
class LocalCacheManager {
  private db: Database; // SQLite数据库
  
  async get(key: string): Promise<any> {
    const result = await this.db.get(
      'SELECT value, expires_at FROM cache WHERE key = ?',
      [key]
    );
    
    if (!result) return null;
    
    // 检查是否过期
    if (result.expires_at && new Date(result.expires_at) < new Date()) {
      await this.delete(key);
      return null;
    }
    
    return JSON.parse(result.value);
  }
  
  async set(key: string, value: any, ttl: number = 3600): Promise<void> {
    const expiresAt = new Date(Date.now() + ttl * 1000);
    await this.db.run(
      'INSERT OR REPLACE INTO cache (key, value, expires_at) VALUES (?, ?, ?)',
      [key, JSON.stringify(value), expiresAt.toISOString()]
    );
  }
}
```

### 3.2 工作流引擎（本地版）
```typescript
// 本地工作流定义
interface LocalWorkflow {
  id: string;
  name: string;
  nodes: WorkflowNode[];
  connections: NodeConnection[];
  variables: Record<string, any>;
  createdAt: Date;
  updatedAt: Date;
  data: string; // 本地文件路径或SQLite BLOB
}

// 工作流执行器
class LocalWorkflowExecutor {
  private workflowStore: LocalWorkflowStore;
  private apiProxy: IMoarkProxyService;
  
  async execute(workflowId: string, inputData: any): Promise<any> {
    // 1. 从本地加载工作流
    const workflow = await this.workflowStore.get(workflowId);
    
    // 2. 创建执行上下文
    const context = {
      workflowId,
      inputData,
      variables: { ...workflow.variables },
      nodeResults: new Map<string, any>(),
      startTime: Date.now()
    };
    
    // 3. 拓扑排序确定执行顺序
    const executionOrder = this.getExecutionOrder(workflow);
    
    // 4. 顺序执行节点
    for (const nodeId of executionOrder) {
      const node = workflow.nodes.find(n => n.id === nodeId);
      if (!node) continue;
      
      try {
        const result = await this.executeNode(node, context);
        context.nodeResults.set(nodeId, result);
      } catch (error) {
        // 错误处理：记录日志，可选继续或停止
        console.error(`节点 ${nodeId} 执行失败:`, error);
        throw error;
      }
    }
    
    // 5. 收集输出
    return this.collectOutputs(workflow, context);
  }
  
  private async executeNode(node: WorkflowNode, context: any): Promise<any> {
    switch (node.type) {
      case 'model':
        return await this.executeModelNode(node, context);
      case 'transform':
        return await this.executeTransformNode(node, context);
      case 'condition':
        return await this.executeConditionNode(node, context);
      case 'loop':
        return await this.executeLoopNode(node, context);
      default:
        throw new Error(`未知节点类型: ${node.type}`);
    }
  }
}
```

```rust
async fn save_api_key(
    app_handle: tauri::AppHandle,

#[tauri::command]
async fn get_usage_stats(
    app_handle: tauri::AppHandle,
    start_date: Option<String>,
    end_date: Option<String>,
) -> Result<serde_json::Value, String> {
    let db_path = get_database_path(&app_handle)?;
    let conn = rusqlite::Connection::open(db_path).map_err(|e| e.to_string())?;
    
    let mut query = "
        SELECT 
            model_id,
            COUNT(*) as call_count,
            SUM(input_tokens) as total_input_tokens,
            SUM(output_tokens) as total_output_tokens,
            SUM(cost) as total_cost,
            AVG(duration_ms) as avg_duration
        FROM model_usage
        WHERE 1=1
    ".to_string();
    
    let mut params: Vec<Box<dyn rusqlite::ToSql>> = vec![];
    
    if let Some(start) = start_date {
        query.push_str(" AND created_at >= ?");
        params.push(Box::new(start));
    }
    
    if let Some(end) = end_date {
        query.push_str(" AND created_at <= ?");
        params.push(Box::new(end));
    }
    
    query.push_str(" GROUP BY model_id ORDER BY call_count DESC");
    
    let mut stmt = conn.prepare(&query).map_err(|e| e.to_string())?;
    let rows = stmt
        .query_map(rusqlite::params_from_iter(params.iter()), |row| {
            Ok(serde_json::json!({
                "model_id": row.get::<_, String>(0)?,
                "call_count": row.get::<_, i64>(1)?,
                "total_input_tokens": row.get::<_, i64>(2)?,
                "total_output_tokens": row.get::<_, i64>(3)?,
                "total_cost": row.get::<_, f64>(4)?,
                "avg_duration": row.get::<_, f64>(5)?,
            }))
        })
        .map_err(|e| e.to_string())?;
    
    let mut stats = Vec::new();
    for row in rows {
        stats.push(row.map_err(|e| e.to_string())?);
    }
    
    Ok(serde_json::json!(stats))
}
```

### 3.4 自动更新系统
```typescript
// 自动更新管理器
class AutoUpdateManager {
  private currentVersion: string;
  private updateUrl: string;
  private checkInterval: number = 24 * 60 * 60 * 1000; // 24小时
  
  constructor() {
    this.currentVersion = process.env.APP_VERSION || '1.0.0';
    this.updateUrl = 'https://api.github.com/repos/your-username/moark-desktop/releases/latest';
  }
  
  async checkForUpdates(): Promise<UpdateInfo | null> {
    try {
      const response = await fetch(this.updateUrl);
      const data = await response.json();
      
      const latestVersion = data.tag_name.replace('v', '');
      
      if (this.isNewerVersion(latestVersion, this.currentVersion)) {
        return {
          version: latestVersion,
          releaseNotes: data.body,
          downloadUrl: data.assets[0]?.browser_download_url,
          publishedAt: new Date(data.published_at),
          isPreRelease: data.prerelease
        };
      }
      
      return null;
    } catch (error) {
      console.error('检查更新失败:', error);
      return null;
    }
  }
  
  async downloadAndInstall(updateInfo: UpdateInfo): Promise<void> {
    // 使用Tauri API下载和安装更新
    const { invoke } = await import('@tauri-apps/api/core');
    
    // 下载更新文件
    const downloadPath = await invoke('download_file', {
      url: updateInfo.downloadUrl,
      filename: `update-${updateInfo.version}.exe`
    });
    
    // 执行安装
    await invoke('execute_installer', {
      path: downloadPath,
      silent: true
    });
    
    // 重启应用
    await invoke('restart_app');
  }
  
  private isNewerVersion(latest: string, current: string): boolean {
    const latestParts = latest.split('.').map(Number);
    const currentParts = current.split('.').map(Number);
    
    for (let i = 0; i < Math.max(latestParts.length, currentParts.length); i++) {
      const latestPart = latestParts[i] || 0;
      const currentPart = currentParts[i] || 0;
      
      if (latestPart > currentPart) return true;
      if (latestPart < currentPart) return false;
    }
    
    return false;
  }
}
```

### 4.2 数据加密方案
```rust
// Rust端数据加密
use aes_gcm::{
    aead::{Aead, KeyInit, OsRng},
    Aes256Gcm, Nonce
};
use rand::RngCore;

pub struct EncryptionManager {
    key: [u8; 32],
}

impl EncryptionManager {
    pub fn new() -> Result<Self, String> {
        // 从安全存储获取或生成主密钥
        let key = Self::get_or_create_master_key()?;
        Ok(Self { key })
    }
    
    pub fn encrypt(&self, plaintext: &str) -> Result<String, String> {
        let cipher = Aes256Gcm::new_from_slice(&self.key)
            .map_err(|e| format!("创建加密器失败: {}", e))?;
        
        let mut nonce = [0u8; 12];
        OsRng.fill_bytes(&mut nonce);
        
        let ciphertext = cipher
            .encrypt(Nonce::from_slice(&nonce), plaintext.as_bytes())
            .map_err(|e| format!("加密失败: {}", e))?;
        
        // 组合 nonce + ciphertext 并Base64编码
        let mut combined = Vec::with_capacity(nonce.len() + ciphertext.len());
        combined.extend_from_slice(&nonce);
        combined.extend_from_slice(&ciphertext);
        
        Ok(base64::encode(combined))
    }
    
    pub fn decrypt(&self, ciphertext: &str) -> Result<String, String> {
        let combined = base64::decode(ciphertext)
            .map_err(|e| format!("Base64解码失败: {}", e))?;
        
        if combined.len() < 12 {
            return Err("密文太短".to_string());
        }
        
        let cipher = Aes256Gcm::new_from_slice(&self.key)
            .map_err(|e| format!("创建解密器失败: {}", e))?;
        
        let nonce = &combined[0..12];
        let ciphertext_bytes = &combined[12..];
        
        let plaintext = cipher
            .decrypt(Nonce::from_slice(nonce), ciphertext_bytes)
            .map_err(|e| format!("解密失败: {}", e))?;
        
        String::from_utf8(plaintext)
            .map_err(|e| format!("UTF-8解码失败: {}", e))
    }
    
    fn get_or_create_master_key() -> Result<[u8; 32], String> {
        // 实现从安全存储获取或生成主密钥的逻辑
        // 可以使用系统密钥链或加密文件
        Ok([0u8; 32]) // 示例，实际需要安全实现
    }
}
```

```tsx
// 工作流编辑器组件
import ReactFlow, {
  Node,
  Edge,
  Controls,
  Background,
  MiniMap,
  useNodesState,
  useEdgesState,
  addEdge,
  Connection,
} from 'reactflow';
import 'reactflow/dist/style.css';

import { ModelNode } from './nodes/ModelNode';
import { InputNode } from './nodes/InputNode';
import { OutputNode } from './nodes/OutputNode';
import { TransformNode } from './nodes/TransformNode';

const nodeTypes = {
  model: ModelNode,
  input: InputNode,
  output: OutputNode,
  transform: TransformNode,
};

export function WorkflowEditor() {
  const [nodes, setNodes, onNodesChange] = useNodesState([]);
  const [edges, setEdges, onEdgesChange] = useEdgesState([]);
  
  const onConnect = (connection: Connection) => {
    setEdges((eds) => addEdge(connection, eds));
  };
  
  const onDragOver = (event: React.DragEvent) => {
    event.preventDefault();
    event.dataTransfer.dropEffect = 'move';
  };
  
  const onDrop = (event: React.DragEvent) => {
    event.preventDefault();
    
    const type = event.dataTransfer.getData('application/reactflow');
    if (!type) return;
    
    const position = {
      x: event.clientX,
      y: event.clientY,
    };
    
    const newNode: Node = {
      id: `${type}_${Date.now()}`,
      type,
      position,
      data: { label: `${type}节点` },
    };
    
    setNodes((nds) => nds.concat(newNode));
  };
  
  return (
    <div className="h-[600px] border rounded-lg">
      <ReactFlow
        nodes={nodes}
        edges={edges}
        onNodesChange={onNodesChange}
        onEdgesChange={onEdgesChange}
        onConnect={onConnect}
        onDrop={onDrop}
        onDragOver={onDragOver}
        nodeTypes={nodeTypes}
        fitView
      >
        <Background />
        <Controls />
        <MiniMap />
      </ReactFlow>
    </div>
  );
}
```

### 7.3 GitHub Actions工作流
```yaml
# .github/workflows/build.yml
name: Build and Release

on:
  push:
    tags:
      - 'v*'

jobs:
  build-tauri:
    strategy:
      fail-fast: false
      matrix:
        platform: [macos-latest, ubuntu-latest, windows-latest]
    
    runs-on: ${{ matrix.platform }}
    
    steps:
    - uses: actions/checkout@v4
    
    - name: Setup Bun
      uses: oven-sh/setup-bun@v1
      with:
        bun-version: latest
    
    - name: Install dependencies
      run: bun install
    
    - name: Build frontend
      run: bun run build
    
    - name: Build Tauri app
      uses: tauri-apps/tauri-action@v0
      env:
        GITHUB_TOKEN: ${{ secrets.GITHUB_TOKEN }}
      with:
        tagName: ${{ github.ref_name }}
        releaseName: 'Moark聚合助手 v${{ github.ref_name }}'
        releaseBody: '查看更新日志: https://github.com/your-username/moark-desktop/releases/tag/${{ github.ref_name }}'
        releaseDraft: true
        prerelease: false
```
